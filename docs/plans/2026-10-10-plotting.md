**Evidence behind the code** (R 4.6 branch source, fetched with `gh api repos/wch/r-source/contents/<path>?ref=R-4-6-branch` into the session scratchpad):

- `stats::cutree` (cutree.R) stops with "invalid 'tree' ('merge' component)" when `nrow(merge) < 1`, then "either 'k' or 'h' must be specified". With `k` it runs `as.integer(k)` and stops with "elements of 'k' must be between 1 and n" outside 1 to n. With only `h` it stops on `is.unsorted(height)` and sets `k <- n + 1 - which.max(c(height, Inf) > h)`.
- `as.integer` on a double gives NA for NaN and for x >= 2^31 or x <= -2^31, and truncates toward zero otherwise (`IntegerFromReal`).
- C `cutree` (hclust-utils.c) keeps `sing` and `m_nr` per row and walks the merges. When `which == n - k` it numbers rows by first appearance. `which == n` gives 1 to n.
- `.validity.hclust` (hclust.R) checks, in order: a two-column matrix ("invalid dendrogram"), integer entries, the height and order lengths, then `sort(as.integer(merge))` against `c(-(n:1), seq_len(n - 2))`.
- `C_dendwindow` and `C_dend` (graphics plot.c) stop with "invalid dendrogram input" for fewer than 2 merges, a non-finite `hang` or labels of the wrong length. `C_dend` scales the hang by `height[n1] - height[1]`, tests the scaled value with `>= 0`, places leaf k at `order(x$order)[k]`, recurses left, right, then draws the four-point polyline and returns its midpoint anchor.
- `r_sum` and `r_quantile7` in `_compat/arithmetic.py` already reproduce R's long double `sum` and type 7 `quantile` bit for bit.

```r
# Fixtures for stats::cutree and stats:::plot.hclust on the trees R's Whclust and DN.Whclust
# returned (read back from the whclust and dn_wcluster groups, recorded on this same R), and for
# the weight colors datanugget's help examples draw nuggets with.
source("r/fixture_io.R")

tree_of <- function(dir) structure(list(merge = read_array(dir, "merge"),
  height = read_array(dir, "height"), order = read_array(dir, "order")), class = "hclust")
stop_message <- function(expr) tryCatch({ force(expr); "" }, error = conditionMessage)

# plot.hclust's polylines, read back from an uncompressed PDF. R's pdf device writes a polyline
# as "x y m", one "x y l" per further point and "S", in big points from the page's bottom left,
# rounded to two decimals. The map from user units to device units is linear in each axis, so
# grconvertX and grconvertY at 0 and 1 invert it.
# A tree with NaN or Inf heights that R doesn't stop on is drawn off the page with a warning, so
# only whether R stopped is recorded for it.
dend_segments <- function(tree, hang = 0.1) {
  f <- tempfile(fileext = ".pdf")
  pdf(f, width = 7, height = 5, compress = FALSE)
  err <- suppressWarnings(
    stop_message(plot(tree, labels = FALSE, axes = FALSE, ann = FALSE, hang = hang)))
  gx <- grconvertX(c(0, 1), "user", "device")
  gy <- grconvertY(c(0, 1), "user", "device")
  dev.off()
  nonfinite <- any(!is.finite(tree$height))
  if (nzchar(err) || nonfinite) { unlink(f); return(list(error = err, nonfinite = nonfinite)) }
  lines <- readLines(f, warn = FALSE)
  unlink(f)
  # The first line is binary, so match bytes, then split the ASCII lines that are left
  lines <- lines[grepl("^(-?[0-9.]+ -?[0-9.]+ [ml]|S)$", lines, useBytes = TRUE)]
  paths <- list(); cur <- NULL
  for (ln in lines) {
    if (ln == "S") {
      paths[[length(paths) + 1L]] <- cur
      cur <- NULL
      next
    }
    p <- strsplit(ln, " ", fixed = TRUE)[[1]]
    cur <- if (p[3] == "m") as.numeric(p[1:2]) else c(cur, as.numeric(p[1:2]))
  }
  n1 <- nrow(tree$merge)
  stopifnot(length(paths) == n1, all(lengths(paths) == 8L))
  segs <- do.call(rbind, paths)
  sx <- gx[2] - gx[1]; sy <- gy[2] - gy[1]
  segs[, c(1, 3, 5, 7)] <- (segs[, c(1, 3, 5, 7)] - gx[1]) / sx
  segs[, c(2, 4, 6, 8)] <- (segs[, c(2, 4, 6, 8)] - gy[1]) / sy
  # Fail closed: the bars sit at the heights, and the leaves span 1 to n
  tol_x <- 0.006 / sx; tol_y <- 0.006 / sy
  stopifnot(max(abs(sort(segs[, 4]) - sort(tree$height))) < tol_y,
            all(abs(segs[, 4] - segs[, 6]) < tol_y),
            abs(min(segs[, c(1, 7)]) - 1) < tol_x, abs(max(segs[, c(1, 7)]) - (n1 + 1)) < tol_x)
  list(error = "", nonfinite = FALSE, segs = segs, scale_x = sx, scale_y = sy)
}

add_tree <- function(id, dir) {
  tree <- tree_of(dir)
  n <- nrow(tree$merge) + 1L
  arrays <- list(merge = tree$merge, height = tree$height, order = tree$order,
                 cut_k = sapply(seq_len(n), function(k) cutree(tree, k = k)))
  hs <- sort(unique(tree$height))
  hs <- c(hs[1] - 1, hs, (hs[-1] + hs[-length(hs)]) / 2, hs[length(hs)] + 1, Inf, -Inf)
  cut_h <- tryCatch(sapply(hs, function(h) cutree(tree, h = h)), error = conditionMessage)
  if (is.character(cut_h)) cut_h_error <- cut_h else {
    cut_h_error <- ""
    arrays <- c(arrays, list(cut_h = hs, cut_h_labels = matrix(as.integer(cut_h), n)))
  }
  d <- dend_segments(tree)
  extra <- list(n = n, plot_error = d$error, nonfinite = d$nonfinite, cut_h_error = cut_h_error)
  if (!is.null(d$segs)) {
    arrays$segs <- d$segs
    extra <- c(extra, list(scale_x = d$scale_x, scale_y = d$scale_y))
  }
  write_case("plot", paste0("tree_", id), no_args, arrays, extra = extra)
  cat(sprintf("plot tree_%-16s n=%3d nonfinite=%s plot=[%s] cut_h=[%s]\n", id, n, d$nonfinite,
              d$error, cut_h_error))
}

for (d in Sys.glob("tests/fixtures/whclust/*")) {
  if ("merge" %in% names(read_json(file.path(d, "manifest.json"))$arrays))
    add_tree(basename(d), d)
}
add_tree("dn", "tests/fixtures/dn_wcluster/whclust")
add_tree("dn_one", "tests/fixtures/dn_wcluster/one_variable")

grid <- tree_of("tests/fixtures/whclust/grid_dup")

write_case("plot", "tree_hang", no_args, list(merge = grid$merge, height = grid$height,
  order = grid$order, segs_m1 = m1$segs, segs_05 = h05$segs),
  extra = list(scale_x = m1$scale_x, scale_y = m1$scale_y))

n3 <- tree_of("tests/fixtures/whclust/n3")
nan_h <- n3; nan_h$height[1] <- NaN
```

```python
"""cut_tree against R's cutree: every k and h R cut, every stop, and the numpy loop against a
statement-for-statement copy of R's C routine on fuzzed trees."""

import numpy as np
import pytest

from datanuggets import WhclustResult, cut_tree
from rfixtures import load_group

CASES = load_group("plot")
TREES = [c for c in CASES if c.id.startswith("tree_") and c.id != "tree_hang"]
STOPS = next(c for c in CASES if c.id == "stops")


def port_tree(case):
    """The recorded R tree in the port's numbering."""
    m = case.array("merge").astype(np.int64)
    n = m.shape[0] + 1
    return WhclustResult(np.where(m < 0, -m - 1, n + m - 1), case.array("height"),
                         case.array("order").astype(np.int64) - 1)


def c_cutree(merge_r, k):
    """R's C cutree for one k, line by line, 1-based like the C."""
    n = len(merge_r) + 1
    sing = [True] * (n + 1)
    m_nr = [0] * (n + 1)
    z = [0] * (n + 1)
    ans = [0] * (n + 1)
    for step in range(1, n):
        m1, m2 = merge_r[step - 1]
        if m1 < 0 and m2 < 0:
            m_nr[-m1] = m_nr[-m2] = step
            sing[-m1] = sing[-m2] = False
        elif m1 < 0 or m2 < 0:
            if m1 < 0:
                j, m1 = -m1, m2
            else:
                j = -m2
            for l in range(1, n + 1):
                if m_nr[l] == m1:
                    m_nr[l] = step
            m_nr[j] = step
            sing[j] = False
        else:
            for l in range(1, n + 1):
                if m_nr[l] == m1 or m_nr[l] == m2:
                    m_nr[l] = step
        if k == n - step:
            z = [0] * (n + 1)
            nclust = 0
            for l in range(1, n + 1):
                if sing[l]:
                    nclust += 1
                    ans[l] = nclust
                else:
                    if z[m_nr[l]] == 0:
                        nclust += 1
                        z[m_nr[l]] = nclust
                    ans[l] = z[m_nr[l]]
    if k == n:
        ans = list(range(n + 1))
    return np.array(ans[1:]) - 1


@pytest.mark.parametrize("case", TREES, ids=[c.id for c in TREES])
def test_every_k_matches_r(case):
    tree = port_tree(case)
    want = case.array("cut_k").astype(np.int64) - 1
    for k in range(1, want.shape[0] + 1):
        assert cut_tree(tree, k=k).tolist() == want[:, k - 1].tolist(), k


@pytest.mark.parametrize("case", TREES, ids=[c.id for c in TREES])
def test_every_h_matches_r(case):
    tree = port_tree(case)
    if case.manifest["cut_h_error"]:
        with pytest.raises(ValueError) as e:
            cut_tree(tree, h=1.0)
        assert str(e.value) == case.manifest["cut_h_error"]
        return
    want = case.array("cut_h_labels").astype(np.int64) - 1
    for col, h in enumerate(case.array("cut_h").tolist()):
        assert cut_tree(tree, h=h).tolist() == want[:, col].tolist(), h


def test_stops_match_r():
    e = STOPS.manifest["errors"]
    tree = port_tree(STOPS)
    nan_h = WhclustResult(tree.merge, [np.nan, tree.height[1]], tree.order)
    unsorted = WhclustResult(tree.merge, tree.height[::-1] + [0, 1], tree.order)
```

```python
"""cut_tree: R's cutree for one k or one h, on a whclust tree."""

from __future__ import annotations

import math

import numpy as np

from datanuggets.result import WhclustResult

# R's message when an if() meets NA: as.integer(NaN), is.unsorted() of NaN heights
NA_STOP = "missing value where TRUE/FALSE needed"
# cutree's h = NaN: which.max of all NA is empty, so k is empty and colnames<- fails
H_NAN_STOP = "length of 'dimnames' [2] not equal to array extent"
_INT_EDGE = 2.0**31  # as.integer gives NA at or beyond +-2^31


def r_merge(merge: np.ndarray) -> np.ndarray:
    """R's merge numbering from the port's: row a below n is -(a + 1), and the cluster merge s
    made is s + 1."""
    m = np.asarray(merge, dtype=np.int64)
    n = m.shape[0] + 1
    return np.where(m < n, -(m + 1), m - n + 1)


def _scalar(v, name: str) -> float:
    if np.ndim(v) != 0:
        raise ValueError(f"cut_tree takes one k or one h, not an array of {name}")
    return float(v)


def _k(k, n: int) -> int:
    """as.integer(k), then cutree's range check."""
    k = _scalar(k, "k")
    if math.isnan(k) or k >= _INT_EDGE or k <= -_INT_EDGE:
        raise ValueError(NA_STOP)
    k = int(k)  # truncates toward zero, as as.integer does
    if k < 1 or k > n:
        raise ValueError(f"elements of 'k' must be between 1 and {n}")
    return k


def _k_from_h(height: np.ndarray, h, n: int) -> int:
    """cutree's k for a height: n + 1 minus the 1-based position of the first entry of
    c(height, Inf) above h."""
    if np.isnan(height).any():
        raise ValueError(NA_STOP)
    if np.any(height[1:] < height[:-1]):
        raise ValueError("the 'height' component of 'tree' is not sorted (increasingly)")
    h = _scalar(h, "h")
    if math.isnan(h):
        raise ValueError(H_NAN_STOP)
    above = np.append(height, np.inf) > h
    return n - int(np.argmax(above))  # argmax of all False is 0, as which.max gives 1


def _cutree(rm: np.ndarray, k: int) -> np.ndarray:
    """R's C cutree for one k on R's merge, 0-based. Each row keeps whether it's still alone and
    the last merge step that took it in. The masks visit the rows R's inner loops visit."""
    n = rm.shape[0] + 1
    if k == n:
        return np.arange(n, dtype=np.int64)
    alone = np.ones(n, dtype=bool)
    step_of = np.zeros(n, dtype=np.int64)
    for step in range(1, n - k + 1):
        m1, m2 = int(rm[step - 1, 0]), int(rm[step - 1, 1])
        if m1 < 0 and m2 < 0:
            step_of[[-m1 - 1, -m2 - 1]] = step
            alone[[-m1 - 1, -m2 - 1]] = False
        elif m1 < 0 or m2 < 0:
            row, other = (-m1 - 1, m2) if m1 < 0 else (-m2 - 1, m1)
            step_of[step_of == other] = step
            step_of[row] = step
            alone[row] = False
        else:
            step_of[(step_of == m1) | (step_of == m2)] = step
    # Number the clusters by first appearance along the rows, a lone row its own cluster
    key = np.where(alone, -np.arange(1, n + 1), step_of)
    _, first, inverse = np.unique(key, return_index=True, return_inverse=True)
    rank = np.empty(first.size, dtype=np.int64)
    rank[np.argsort(first)] = np.arange(first.size)
    return rank[inverse.ravel()]


def cut_tree(tree: WhclustResult, k=None, h=None) -> np.ndarray:
    """R's cutree(tree, k, h): each row's cluster when the tree is cut into k clusters, or at
    height h, as R's label minus one. k wins when both are given, and a fractional k is
    truncated as R's as.integer does. Raises ValueError with R's message where R stops."""
    n1 = tree.merge.shape[0]
    if n1 < 1:
        raise ValueError("invalid 'tree' ('merge' component)")
    n = n1 + 1
    if k is None and h is None:
        raise ValueError("either 'k' or 'h' must be specified")
    k = _k(k, n) if k is not None else _k_from_h(np.asarray(tree.height, dtype=float), h, n)
    return _cutree(r_merge(tree.merge), k)
```

```python
"""dendrogram: R's plot.hclust for a whclust tree, line for line."""

from __future__ import annotations

import math

import matplotlib.pyplot as plt
import numpy as np
from matplotlib.collections import LineCollection

from datanuggets.wcluster.cutree import r_merge

BAD_INPUT = "invalid dendrogram input"  # C_dend's and C_dendwindow's one message
# .validity.hclust's seq_len(n - 2) for a tree with no merges
NO_MERGE_STOP = "argument must be coercible to non-negative integer"
# The port's own: R warns and draws such a tree off the page (README departure 5)
NONFINITE_STOP = "the tree has NaN or infinite heights, which R's plot.hclust draws off the page"


def _r_merge_checked(tree) -> np.ndarray:
    """.validity.hclust's checks in R's order, then C_dendwindow's: R's merge, or ValueError
    with R's message."""
    merge = np.asarray(tree.merge)
    if merge.ndim != 2 or merge.shape[1] != 2:
        raise ValueError("invalid dendrogram")
    if not np.issubdtype(merge.dtype, np.integer) and not np.array_equal(merge, np.trunc(merge)):
        raise ValueError("'merge' component in dendrogram must be integer")
    n1 = merge.shape[0]
    n = n1 + 1
    if np.size(tree.height) != n1:
        raise ValueError("'height' is of wrong length")
    if np.size(tree.order) != n:
        raise ValueError("'order' is of wrong length")
    if n1 == 0:
        raise ValueError(NO_MERGE_STOP)
    rm = r_merge(merge.astype(np.int64))
    want = np.concatenate([np.arange(-n, 0), np.arange(1, n - 1)])
    if not np.array_equal(np.sort(rm, axis=None), want):
        raise ValueError("'merge' matrix has invalid contents")
    if n1 < 2:
        raise ValueError(BAD_INPUT)
    return rm


def _check_heights(height: np.ndarray) -> None:
    """Whclust gives NaN heights for zero weights and Inf ones for infinite weights. R's
    plot.hclust only warns on them and writes coordinates no figure can show, so the port stops."""
    if not np.isfinite(height).all():
        raise ValueError(NONFINITE_STOP)


def _drawdend(rm, height, xpos, hang):
    """C_dend's drawdend from the last merge, as an explicit stack: the left subtree, the right
    subtree, then the merge's polyline. A leaf hangs hang below its merge, or sits at 0 when
    hang is negative. Returns the polylines in R's drawing order and each leaf's (row, x, y)."""
    lines, leaves, anchor = [], [], {}
    stack = [(rm.shape[0], False)]
    while stack:
        node, ready = stack.pop()
        left, right = int(rm[node - 1, 0]), int(rm[node - 1, 1])
        if not ready:
            stack.append((node, True))
            stack.extend((k, False) for k in (right, left) if k > 0)  # left pops first
            continue
        y = float(height[node - 1])
        ends = []
        for k in (left, right):
            if k > 0:
                ends.append(anchor.pop(k))
            else:
                end = (float(xpos[-k - 1]), y - hang if hang >= 0 else 0.0)
                leaves.append((-k - 1, *end))
                ends.append(end)
        (xl, yl), (xr, yr) = ends
        lines.append(((xl, yl), (xl, y), (xr, y), (xr, yr)))
        anchor[node] = (0.5 * (xl + xr), y)
    return np.array(lines), leaves


def _leaf_text(labels, n: int):
    if labels is None:
        return [str(i) for i in range(n)]
    if labels is False:
        return None
    text = [str(v) for v in labels]
    if len(text) != n:
        raise ValueError(BAD_INPUT)
    return text


def dendrogram(tree, *, labels=None, hang=0.1, ax=None):
    """Draws a whclust tree as R's plot(tree) does: leaf i at x = 1 + its place in tree.order,
    each merge a bracket at its height, leaves hanging hang times the last height minus the
    first below their merge, or at 0 when that's negative. labels are the 0-based row numbers by
    default, a sequence of n strings, or False for none. Only the y axis is drawn, labelled
    "Height". Raises ValueError with R's message where plot.hclust stops, such as for a tree
    with a self-merge, and for a tree with NaN or infinite heights, which R draws off the page.
    Returns the Axes."""
    rm = _r_merge_checked(tree)
    n = rm.shape[0] + 1
    hang = float(hang)
    if not math.isfinite(hang):
        raise ValueError(BAD_INPUT)
    text = _leaf_text(labels, n)
    height = np.asarray(tree.height, dtype=float)
    _check_heights(height)
    xpos = np.argsort(np.asarray(tree.order), kind="stable") + 1.0  # R's order(x$order)
    lines, leaves = _drawdend(rm, height, xpos, hang * (height[-1] - height[0]))
    ax = plt.gca() if ax is None else ax
    rc = plt.rcParams
    ax.add_collection(LineCollection(lines, colors=rc["axes.edgecolor"],
                                     linewidths=rc["lines.linewidth"], gid="dendrogram"))
    ax.autoscale_view()
    if text is not None:
        for row, x, y in leaves:
            ax.annotate(text[row], (x, y), xytext=(0, -2), textcoords="offset points",
                        rotation=90, ha="center", va="top", fontsize=rc["xtick.labelsize"],
                        annotation_clip=False)
    ax.set_ylabel("Height")
    ax.xaxis.set_visible(False)
    for side in ("top", "right", "bottom"):
        ax.spines[side].set_visible(False)
    return ax
```

`_check_heights` runs after every check R makes, so `selfmerge_mid`, which has an Inf height and a wrong order length, gets R's message.

Then in `plot/__init__.py`: `from datanuggets.plot.tree import dendrogram` and `"dendrogram"` in `__all__`.

```python
"""clusters and nuggets: the scatter plots of WCluster's and datanugget's help examples."""

from __future__ import annotations

import numpy as np
import matplotlib.pyplot as plt

from datanuggets._compat.arithmetic import r_quantile7, r_sum

PCH16 = 9.0  # R's pch = 16 at cex 1: a disc 0.375 * 12 pt in radius, so 9 pt across
NUGGET_CEX = 0.5  # the help examples' nugget size
CENTER_SIZE = 120.0  # star markers for cluster centers, points squared
PLAIN_SIZE = 20.0  # a point without a weight
# quantile's stop when the weight fractions have a NaN, from 0 / 0 or an infinite weight
QUANTILE_NAN_STOP = "missing values and NaN's not allowed if 'na.rm' is FALSE"


def _matrix(x) -> np.ndarray:
    a = np.asarray(x, dtype=float)
    if a.ndim != 2:
        raise ValueError(f"expected a two-dimensional array, got {a.ndim} dimensions")
    return a


def _dims(dims, p: int) -> tuple[int, int]:
    d = tuple(int(v) for v in dims)
    if len(d) != 2 or d[0] == d[1] or not all(0 <= v < p for v in d):
        raise ValueError(f"dims must be two distinct columns from 0 to {p - 1}, got {dims}")
    return d


def _labels(labels, n: int) -> np.ndarray:
    a = np.asarray(labels)
    if a.shape != (n,) or not np.issubdtype(a.dtype, np.integer):
        raise ValueError(f"labels must be {n} integers, one per point")
    return a


def _palette(colors) -> list:
    if colors is None:
        return plt.rcParams["axes.prop_cycle"].by_key()["color"]
    return list(colors)


def _label_colors(labels, palette) -> list:
    return [palette[int(v) % len(palette)] for v in labels]


def _sizes(w, n: int, size: str) -> np.ndarray:
    """Marker areas in points squared: proportional to weight ("area", 6 to 406), or R's
    cex = log(w) for pch = 16 ("log"), where a log that isn't positive draws nothing."""
    if size not in ("area", "log"):
        raise ValueError(f"size must be 'area' or 'log', got {size!r}")
    if w is None:
        if size == "log":
            raise ValueError("size='log' needs w")
        return np.full(n, PLAIN_SIZE)
    w = np.asarray(w, dtype=float)
    if w.shape != (n,):
        raise ValueError(f"w has {w.size} entries for {n} points")
    if size == "log":
        with np.errstate(divide="ignore", invalid="ignore"):
            cex = np.log(w)
        return np.where(cex > 0, (PCH16 * cex) ** 2, 0.0)
    if not (np.all(np.isfinite(w)) and np.all(w >= 0) and w.max() > 0):
        raise ValueError("size='area' needs finite weights, none negative and not all zero")
    return 6.0 + 400.0 * w / w.max()


def _weight_fractions(w) -> np.ndarray:
    """The examples' t: w / sum(w), then over its 80% quantile, with R's long double sum and
    type 7 quantile."""
    w = np.asarray(w, dtype=float)
    with np.errstate(divide="ignore", invalid="ignore"):
        t = w / r_sum(w)
        if np.isnan(t).any():
            raise ValueError(QUANTILE_NAN_STOP)
        return t / r_quantile7(t, 0.8)


def _rgb_stop(v: float) -> str:
    """ScaleColor's message, which prints the value with %g (NaN as "nan")."""
    return f"color intensity {v:g}, not in [0,1]"


def _rgb_green(v: float) -> str:
    """R's rgb(0, v, 0): ScaleColor's checks, then 255 v + 0.5 truncated."""
    if not (v >= 0 and v <= 1):
        raise ValueError(_rgb_stop(v))
    return f"#00{int(255 * v + 0.5):02X}00"


def weight_colors(w) -> list[str]:
    """The help examples' nugget colors: the fraction of weight over its 80% quantile, capped at
    1, as rgb(0, t, 0). Lighter green means more weight. The hex strings are R's."""
    return [_rgb_green(v) for v in np.minimum(_weight_fractions(w), 1.0).tolist()]


def clusters(x, labels, *, w=None, centers=None, dims=(0, 1), size="area", colors=None,
             ax=None):
    """The Whclust and Wkmeans examples' plot(x, col = labels) with points(centers): the rows of
    x on columns dims, label l in colors[l % len(colors)] (the color cycle by default, R_PALETTE
    for R's), sized by w (see size), and centers, one row per label present in label order, as
    stars in their cluster's color. Returns the Axes."""
    x = _matrix(x)
    n, p = x.shape
    i, j = _dims(dims, p)
    labels = _labels(labels, n)
    s = _sizes(w, n, size)
    palette = _palette(colors)
    ax = plt.gca() if ax is None else ax
    ax.scatter(x[:, i], x[:, j], s=s, c=_label_colors(labels, palette), linewidths=0,
               gid="points")
    if centers is not None:
        c = _matrix(centers)
        present = np.unique(labels)
        if c.shape != (present.size, p):
            raise ValueError(f"centers must be {present.size} by {p}, one row per label present")
        ax.scatter(c[:, i], c[:, j], s=CENTER_SIZE, marker="*", c=_label_colors(present, palette),
                   edgecolors="black", linewidths=0.6, zorder=3, gid="centers")
    return ax


def nuggets(dn, *, labels=None, dims=(0, 1), colors=None, ax=None):
    """The DN examples' plot of nugget centers on columns dims, pch = 16 at cex = 0.5, colored
    by weight as R's examples do (weight_colors), or by labels like clusters. Returns the
    Axes."""
    c = np.asarray(dn.centers, dtype=float)
    m, p = c.shape
    i, j = _dims(dims, p)
    if labels is None:
        face = weight_colors(dn.weights)
    else:
        face = _label_colors(_labels(labels, m), _palette(colors))
    ax = plt.gca() if ax is None else ax
    ax.scatter(c[:, i], c[:, j], s=(PCH16 * NUGGET_CEX) ** 2, c=face, linewidths=0,
               gid="nuggets")
    return ax
```

`all_zero` reaches the NaN check (0 / 0). `quantile_zero` reaches `rgb` with NaN (0 / 0 for the zero weights), and `np.minimum` keeps the NaN as R's `min(t, 1)` does. The colors are made in row order, so the first bad value names the error, as in R's `sapply`.

```python
"""Weighted PCA views: scores, the scree and the loadings, with each component's sign fixed so
its largest loading is positive. WpcaResult keeps R's signs, only the views flip."""

from __future__ import annotations

import math

import matplotlib.pyplot as plt
import numpy as np

from datanuggets.plot.points import _dims, _label_colors, _matrix, _palette, _sizes

MAX_ROWS = 50_000  # raw rows drawn behind the scores, every ceil(n / MAX_ROWS)-th


def _signs(rotation) -> np.ndarray:
    """+1 or -1 per component: -1 where the loading largest by absolute value is negative."""
    r = np.asarray(rotation, dtype=float)
    big = r[np.argmax(np.abs(r), axis=0), np.arange(r.shape[1])]
    return np.where(big < 0, -1.0, 1.0)


def _share(wpca) -> np.ndarray:
    """Each component's percent of the variance. An eigenvalue rounded just below zero gives a
    NaN sdev, counted as 0."""
    lam = np.nan_to_num(np.asarray(wpca.sdev, dtype=float) ** 2)
    return 100.0 * lam / lam.sum()


def _plane(wpca, dims) -> tuple[int, int]:
    q = np.asarray(wpca.rotation).shape[1]
    if q < 2:
        raise ValueError("a PCA plane needs at least two components")
    return _dims(dims, q)


def scores(wpca, *, w=None, labels=None, rows=None, dims=(0, 1), colors=None, ax=None):
    """The scores on components dims, sized by w (area proportional to weight) and colored by
    labels when given. rows, raw data, are drawn behind in light gray, projected as the scores
    were: (rows - center) @ rotation, at most 50,000 of them. The axis labels give each
    component's share of the variance. Returns the Axes."""
    i, j = _plane(wpca, dims)
    s = _signs(wpca.rotation)
    x = np.asarray(wpca.x, dtype=float) * s
    n = x.shape[0]
    ax = plt.gca() if ax is None else ax
    if rows is not None:
        r = _matrix(rows)
        if r.shape[1] != np.size(wpca.center):
            raise ValueError(f"rows must have {np.size(wpca.center)} columns")
        r = r[:: max(1, math.ceil(r.shape[0] / MAX_ROWS))]
        proj = (r - wpca.center) @ (np.asarray(wpca.rotation, dtype=float) * s)[:, [i, j]]
        ax.scatter(proj[:, 0], proj[:, 1], s=2, c="0.8", linewidths=0, rasterized=True,
                   gid="rows")
    palette = _palette(colors)
    face = palette[0] if labels is None else _label_colors(_labels_for(labels, n), palette)
    ax.scatter(x[:, i], x[:, j], s=_sizes(w, n, "area"), c=face, linewidths=0, gid="scores")
    share = _share(wpca)
    ax.set_xlabel(f"PC{i + 1} ({share[i]:.1f}%)")
    ax.set_ylabel(f"PC{j + 1} ({share[j]:.1f}%)")
    return ax


def scree(wpca, *, ax=None):
    """Each component's percent of the variance as a bar. Returns the Axes."""
    share = _share(wpca)
    ax = plt.gca() if ax is None else ax
    k = np.arange(1, share.size + 1)
    ax.bar(k, share, color=_palette(None)[0], gid="scree")
    ax.set_xticks(k)
    ax.set_xlabel("Component")
    ax.set_ylabel("Variance explained (%)")
    return ax


def loadings(wpca, *, names=None, dims=(0, 1), ax=None):
    """An arrow from the origin to each variable's loadings on components dims, labelled with
    names or the 0-based column numbers. Returns the Axes."""
    i, j = _plane(wpca, dims)
    rot = np.asarray(wpca.rotation, dtype=float) * _signs(wpca.rotation)
    p = rot.shape[0]
    text = [str(v) for v in (range(p) if names is None else names)]
    if len(text) != p:
        raise ValueError(f"names must have {p} entries, one per variable")
    ax = plt.gca() if ax is None else ax
    ax.axhline(0, color="0.85", linewidth=0.5, zorder=0)
    ax.axvline(0, color="0.85", linewidth=0.5, zorder=0)
    ax.quiver(np.zeros(p), np.zeros(p), rot[:, i], rot[:, j], angles="xy", scale_units="xy",
              scale=1, width=0.005, gid="loadings")
    for k in range(p):
        ax.text(1.1 * rot[k, i], 1.1 * rot[k, j], text[k], ha="center", va="center")
    ax.set_xlim(-1.2, 1.2)
    ax.set_ylim(-1.2, 1.2)
    ax.set_aspect("equal")
    ax.set_xlabel(f"PC{i + 1}")
    ax.set_ylabel(f"PC{j + 1}")
    return ax
```

`_labels_for` is `points._labels`, imported under that name. Then in `plot/__init__.py`: `from datanuggets.plot.pca import loadings, scores, scree` and the three names in `__all__`.
