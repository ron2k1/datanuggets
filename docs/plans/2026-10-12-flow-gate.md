```python
"""His lymphmacro_finder cut at its steps, each step his own lines, so each can be timed and
recorded on its own. finder() chains the steps, and record.py and check_reference.py check that it
gives his lymphmacro_finder's masks. graph() is his plotting block without plt.show()."""

import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
from cabrera_reference import f_plot, find_area, search_ind
from scipy.ndimage import generic_filter
from scipy.spatial import ConvexHull, Delaunay


def transform(fsc_a, ssc_a):
    min_val = min(np.min(ssc_a), np.min(fsc_a))
    x0 = np.sqrt((-min_val) + 3 + ssc_a)
    y0 = np.sqrt((-min_val) + 3 + fsc_a)
    return x0, y0


def grid(x0, y0):
    msize = 100
    return f_plot(x0, y0, nr=msize, nc=msize, scale="l")


def maxima(f_density, n_row=11, n_col=11):
    x_matrix = f_density["z"].T
    threshold_val = np.quantile(f_density["z"], 0.85)

    def focal_func(window):
        m = np.max(window)
        return m if m > threshold_val else -1

    local_max = generic_filter(x_matrix, focal_func, size=(n_row, n_col))
    max_coords = np.argwhere(x_matrix == local_max)
    return np.column_stack([max_coords[:, 0] + 1, max_coords[:, 1] + 1])


def peaks(x_matrix, maxXY_rot):
    idx_f = []
    for pt in maxXY_rot:
        idx_f.append(search_ind(x_matrix, pt[0], pt[1]))
    idx_f = np.unique(np.array(idx_f), axis=0)
    mask = (idx_f[:, 0] < 90) & (idx_f[:, 1] < 90) & (idx_f[:, 0] > 10)
    idx_f = idx_f[mask]
    q90 = np.quantile(x_matrix, 0.9)
    idx_f = idx_f[np.array([x_matrix[int(i) - 1, int(j) - 1] for i, j in idx_f]) > q90]
    return idx_f[np.argsort(idx_f[:, 0])]


def areas(x_matrix, idx_f, t11=22, t21=0.25, t12=20, t22=0.5):
    idx_n = len(idx_f)
    tt = np.zeros((2, idx_n))
    tt[0, 0], tt[1, 0] = t11, t21
    tt[0, 1:], tt[1, 1:] = t12, t22
    dat_list = []
    for i in range(idx_n):
        area = find_area(x_matrix, idx_f[i, 0], idx_f[i, 1], t1=tt[0, i], t2=tt[1, i])
        area_with_id = np.column_stack([area, np.full(len(area), i + 1)])
        dat_list.append(area_with_id)
    return pd.DataFrame(np.vstack(dat_list), columns=["x", "z", "y"])


def gate(fsc_a, ssc_a, f_density, dat, idx_n):
    raw_data = np.column_stack([fsc_a, ssc_a])
    zz_grid = f_density["zz"]
    final_indices = np.zeros((len(raw_data), idx_n), dtype=bool)
    for i in range(1, idx_n + 1):
        v_cluster = dat[dat["y"] == i][["x", "z"]].values
        if len(v_cluster) > 3:
            hull = ConvexHull(v_cluster)
            delaunay = Delaunay(v_cluster[hull.vertices])
            jj_type = delaunay.find_simplex(zz_grid) >= 0
            points_to_gate = raw_data[jj_type]
            if len(points_to_gate) > 3:
                data_hull = Delaunay(points_to_gate)
                final_indices[:, i - 1] = data_hull.find_simplex(raw_data) >= 0
    return final_indices


def finder(fsc_a, ssc_a, n_row=11, n_col=11, t11=22, t21=0.25, t12=20, t22=0.5):
    x0, y0 = transform(fsc_a, ssc_a)
    f_density = grid(x0, y0)
    x_matrix = f_density["z"].T
    idx_f = peaks(x_matrix, maxima(f_density, n_row, n_col))
    dat = areas(x_matrix, idx_f, t11, t21, t12, t22)
    return gate(fsc_a, ssc_a, f_density, dat, len(idx_f))


def graph(a, ind, title):
    fig = plt.figure(figsize=(6, 6))
    plt.scatter(a[:, 0], a[:, 1], s=0.5, c="gray", alpha=0.3, label="Unlabeled")
    num_pops = ind.shape[1]
    for j in range(num_pops):
        gated_mask = ind[:, j]
        if np.any(gated_mask):
            plt.scatter(a[gated_mask, 0], a[gated_mask, 1], s=5, label=f"Pop {j + 1}", alpha=0.8)
    plt.title(title)
    plt.xlabel("FSC-A")
    plt.ylabel("SSC-A")
    plt.legend(markerscale=4)
    plt.tight_layout()
    return fig
```

```python
"""The benchmark's FCS reading: FSC-A and SSC-A as FlowKit's raw events hold them, and the files
of his study layout."""

from pathlib import Path

import flowio
import numpy as np
import pandas as pd


def read_fsc_ssc(path) -> tuple[np.ndarray, np.ndarray]:
    """FSC-A and SSC-A as float64. FlowKit's Sample.get_events(source="raw") is flowio's
    as_array(preprocess=True), which the mini showed equal on three of these files."""
    fd = flowio.FlowData(str(path))
    events = fd.as_array(preprocess=True)
    names = [fd.text[f"p{k}n"] for k in range(1, fd.channel_count + 1)]
    return events[:, names.index("FSC-A")], events[:, names.index("SSC-A")]


def fcs_files(target) -> list[Path]:
    """One FCS file, or every *.fcs in a folder, sorted by name."""
    p = Path(target).expanduser()
    files = sorted(p.glob("*.fcs")) if p.is_dir() else [p]
    if not files or not all(f.is_file() for f in files):
        raise FileNotFoundError(f"no FCS file at {p}")
    return files


def study_files(metadata, folder) -> list[Path]:
    """The files his script reads: the Filename column of the workbook's second sheet."""
    dt_meta = pd.read_excel(Path(metadata).expanduser(), sheet_name=1)
    return [Path(folder).expanduser() / f for f in dt_meta["Filename"]]
```

- [ ] **Step 5: `prepare.py`**

```python
"""Put a folder of FCS files in Cabrera's study layout: a workbook whose second sheet has the
Filename column his script reads. With --source, the Zenodo sheet's columns come along.

usage: python benchmarks/flow/prepare.py <folder> [--source metadata_upload.xlsx] [--limit N]
           [--out <folder>/study_metadata.xlsx]
"""

import argparse
from pathlib import Path

import pandas as pd
from fcs import fcs_files


def main() -> None:
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    ap.add_argument("folder")
    ap.add_argument("--source", help="the record's metadata sheet, with a file_name column")
    ap.add_argument("--limit", type=int, help="keep the first N files")
    ap.add_argument("--out")
    args = ap.parse_args()
    folder = Path(args.folder).expanduser()
    names = [f.name for f in fcs_files(folder)]
    samples = pd.DataFrame({"Filename": names})
    if args.source:
        meta = pd.read_excel(Path(args.source).expanduser()).rename(columns={"file_name": "Filename"})
        samples = samples.merge(meta, on="Filename", how="left")
    if args.limit:
        samples = samples.head(args.limit)
    out = Path(args.out).expanduser() if args.out else folder / "study_metadata.xlsx"
    study = pd.DataFrame({"Study": ["Zenodo 22079206, PBMC of critically ill patients, CC-BY-4.0"]})
    with pd.ExcelWriter(out) as xw:
        study.to_excel(xw, sheet_name="Study", index=False)
        samples.to_excel(xw, sheet_name="Samples", index=False)
    print(f"wrote {out}: {len(samples)} files on sheet 2")


if __name__ == "__main__":
    main()
```

```python
@dataclass(frozen=True)
class FlowCase:
    name: str
    fsc: np.ndarray
    ssc: np.ndarray
    z: np.ndarray
    gx: np.ndarray
    gy: np.ndarray
    zz: np.ndarray

@cache

        n = len(f["fsc"])

        def unpack(key: str) -> np.ndarray:
            return np.unpackbits(f[key], axis=0, count=n).astype(bool)

        def ints(key: str) -> np.ndarray:
            return f[key].astype(np.int64)

            peaks=ints("peaks"), his_area=ints("his_area"), fixed_area=ints("fixed_area"),
            his_masks=unpack("his_masks"), fixed_masks=unpack("fixed_masks"),
        )


def mirrored_area(x, idx, idy, t1, t2):
    """What his find_area keeps, stated directly: each cell's own density, the disc centred on the
    mirrored peak (idy, idx). Cells as (row, col), column-major."""
    a, b = np.indices(x.shape) + 1
    keep = (np.hypot(a - idy, b - idx) < t1) & (x > t2 * x[idx - 1, idy - 1])
    return np.column_stack((a.T[keep.T], b.T[keep.T]))


def fixed_area(x, idx, idy, t1, t2):
    """The corrected rule stated directly: the same with the disc on the peak."""
    a, b = np.indices(x.shape) + 1
    keep = (np.hypot(a - idx, b - idy) < t1) & (x > t2 * x[idx - 1, idy - 1])
    return np.column_stack((a.T[keep.T], b.T[keep.T]))


def edge_distance(points: np.ndarray, q: np.ndarray) -> np.ndarray:
    """Each row of q's distance to the boundary of the convex hull of points."""
    v = points[ConvexHull(points).vertices]
    d = np.roll(v, -1, axis=0) - v
    rel = q[:, None, :] - v[None]
    t = np.clip((rel * d[None]).sum(-1) / (d * d).sum(-1), 0.0, 1.0)
    return np.sqrt(((rel - t[..., None] * d[None]) ** 2).sum(-1)).min(axis=1)


def assert_gate_close(fsc, ssc, got, want, zz, cells) -> None:
    """Spec criterion 3: the masks are equal, or every event that differs lies within EDGE_TOL of
    the data's range from the edge of the hull of the events whose bin is in the cells' hull."""
    from datanuggets.flow.gate import bin_mask

    if np.array_equal(got, want):
        return
    raw = np.column_stack((fsc, ssc))
    diff = np.flatnonzero(got != want)
    span = max(np.ptp(fsc), np.ptp(ssc))
    far = edge_distance(raw[bin_mask(zz, cells)], raw[diff]).max()
    assert far <= EDGE_TOL * span, f"{len(diff)} events differ, the farthest {far:.3g} from the edge"
```

```python
"""f_plot: Cabrera's density grid of two channels, his bins counted with one bincount."""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np

from datanuggets._validate import as_count, as_real, check_choice

SCALES = ("raw", "l", "d")  # his counts, log(1 + z), log(1 + log(1 + z))


@dataclass(frozen=True)
class DensityGrid:
    """His f_plot's result. z is the nc by nr grid (his t(z)), x and y the nr and nc evenly spaced
    positions from each channel's minimum to its maximum, zz each event's 1-based (x bin, y bin)."""

    z: np.ndarray
    x: np.ndarray
    y: np.ndarray
    zz: np.ndarray


def channel(name: str, v) -> np.ndarray:
    """v as a float64 vector his binning can use: finite, with more than one value."""
    a = np.asarray(v, dtype=float)
    if a.ndim != 1 or a.size == 0:
        raise ValueError(f"{name} must be a non-empty one-dimensional array")
    if not np.isfinite(a).all():
        raise ValueError(f"{name} has NaN or infinite values")
    if a.min() == a.max():
        raise ValueError(f"{name} has one value, so it can't be binned")
    return a


def sqrt_coords(fsc_a, ssc_a) -> tuple[np.ndarray, np.ndarray]:
    """His transform: SSC-A and FSC-A shifted by the smaller channel's minimum, plus 3, square
    rooted. x is SSC-A, y is FSC-A, as he wrote it."""
    min_val = min(np.min(ssc_a), np.min(fsc_a))
    return np.sqrt((-min_val) + 3 + ssc_a), np.sqrt((-min_val) + 3 + fsc_a)


def _bins(v: np.ndarray, n: int) -> np.ndarray:
    """His bin of each value, floor(n (v - min) / (max - min)) + 1 clipped to 1 to n, in his
    order of operations so every bin is his."""
    lo, hi = v.min(), v.max()
    return np.clip(np.floor(n * (v - lo) / (hi - lo)).astype(int) + 1, 1, n)


def f_plot(x, y, *, nr=20, nc=20, scale="raw", alpha=0.1) -> DensityGrid:
    """His f_plot: the nr by nc table of the binned events, scaled, with cells below alpha times
    the maximum set to 0. His padding for R's table() isn't needed by bincount, so it's gone."""
    x, y = channel("x", x), channel("y", y)
    if len(x) != len(y):
        raise ValueError(f"x and y must have the same length, got {len(x)} and {len(y)}")
    nr, nc = as_count("nr", nr), as_count("nc", nc)
    check_choice("scale", scale, SCALES)
    alpha = as_real("alpha", alpha)
    zx, zy = _bins(x, nr), _bins(y, nc)
    z = np.bincount((zx - 1) * nc + (zy - 1), minlength=nr * nc).reshape(nr, nc).astype(float)
    if scale == "l":
        z = np.log1p(z)
    elif scale == "d":
        z = np.log1p(np.log1p(z))
    z[z < z.max() * alpha] = 0
    return DensityGrid(
        z=z.T, x=np.linspace(x.min(), x.max(), nr), y=np.linspace(y.min(), y.max(), nc),
        zz=np.column_stack((zx, zy)),
    )
```

```python
"""The peaks of the density grid: his focal maxima, refined by search_ind and filtered the way he
filters them, and find_area, the cells around each."""

from __future__ import annotations

import math

import numpy as np
from scipy.ndimage import maximum_filter

ROW_MIN = 10  # his filters on a refined peak: row > 10, row < 90, column < 90
ROW_MAX = 90
COL_MAX = 90
PEAK_QUANTILE = 0.9  # a peak's density must be above this quantile of the grid


def local_maxima(z, *, n_row=11, n_col=11, quantile=0.85) -> np.ndarray:
    """His focal step: the 1-based (row, col) of every cell equal to the maximum of the n_row by
    n_col window around it, when that maximum is above the quantile of z. maximum_filter's reflect
    edges are generic_filter's, so the windows are his."""
    z = np.asarray(z, dtype=float)
    m = maximum_filter(z, size=(n_row, n_col), mode="reflect")
    return np.argwhere(z == np.where(m > np.quantile(z, quantile), m, -1.0)) + 1


def search_ind(x, idx, idy, threshold=2) -> np.ndarray:
    """The 1-based cell of the largest x in the window of half width threshold around (idx, idy),
    clipped to the grid, the first in row order on a tie, as his argmax takes it."""
    r, c = math.ceil(idx), math.ceil(idy)
    r0, r1 = max(r - threshold, 1), min(r + threshold, x.shape[0])
    c0, c1 = max(c - threshold, 1), min(c + threshold, x.shape[1])
    w = x[r0 - 1 : r1, c0 - 1 : c1]
    i, j = np.unravel_index(np.argmax(w), w.shape)
    return np.array([r0 + i, c0 + j])


def refine_peaks(x, candidates, *, threshold=2) -> np.ndarray:
    """His peak step: search_ind from each candidate, np.unique, his row and column limits, a
    density above the 0.9 quantile, then np.argsort on the row. The same calls in his order,
    since unique's sort and argsort decide which peak comes first and gets t11 and t21."""
    if len(candidates) == 0:
        return np.empty((0, 2), dtype=np.int64)
    p = np.unique(np.array([search_ind(x, i, j, threshold) for i, j in candidates]), axis=0)
    p = p[(p[:, 0] < ROW_MAX) & (p[:, 1] < COL_MAX) & (p[:, 0] > ROW_MIN)]
    p = p[x[p[:, 0] - 1, p[:, 1] - 1] > np.quantile(x, PEAK_QUANTILE)]
    return p[np.argsort(p[:, 0])]
```

```python
def find_area(x, idx, idy, t1=35, t2=0.85) -> np.ndarray:
    """The 1-based cells (i, j) of x closer than t1 to the peak (idx, idy) with x[i, j] above t2
    times the peak's density, column by column as his come. His version reads each cell's
    density right but centres the distance on the mirrored peak (idy, idx), so off the diagonal
    it keeps the wrong cells (spec section 4, piece 4). This one centres it on the peak."""
    x = np.asarray(x, dtype=float)
    di = np.arange(1, x.shape[0] + 1)[:, None] - idx
    dj = np.arange(1, x.shape[1] + 1)[None, :] - idy
    keep = (np.sqrt(di * di + dj * dj) < t1) & (x > t2 * x[int(idx) - 1, int(idy) - 1])
    j, i = np.nonzero(keep.T)
    return np.column_stack((i, j)) + 1
```

```python
"""hull_gate and lymphmacro_finder: the events of each population, in the convex hull of the
events whose bin lies in the hull of the population's area."""

from __future__ import annotations

import numpy as np
from scipy.spatial import ConvexHull, Delaunay, QhullError

from datanuggets.flow.grid import channel, f_plot, sqrt_coords
from datanuggets.flow.peaks import find_area, local_maxima, refine_peaks

MSIZE = 100  # his grid, 100 by 100 bins
MIN_POINTS = 3  # his guard: a hull needs more than 3 cells, and more than 3 kept events


def _inside(vertices: np.ndarray, q: np.ndarray) -> np.ndarray:
    """His test, Delaunay(...).find_simplex(q) >= 0, on a triangulation of the hull's vertices."""
    return Delaunay(vertices).find_simplex(q) >= 0


def bin_mask(zz, cells) -> np.ndarray:
    """The events whose bin is inside the convex hull of the cells. Each grid cell is tested
    once and looked up per event, since every event in a bin has the bin's coordinates."""
    zz = np.asarray(zz)
    cells = np.asarray(cells)
    if len(cells) <= MIN_POINTS:
        return np.zeros(len(zz), dtype=bool)
    try:
        hull = ConvexHull(cells)
    except QhullError:  # cells on one line: his ConvexHull raises here
        return np.zeros(len(zz), dtype=bool)
    nr, nc = int(zz[:, 0].max()), int(zz[:, 1].max())
    a, b = np.indices((nr, nc)) + 1
    inside = _inside(cells[hull.vertices], np.column_stack((a.ravel(), b.ravel())))
    return inside.reshape(nr, nc)[zz[:, 0] - 1, zz[:, 1] - 1]


def hull_gate(fsc_a, ssc_a, zz, cells) -> np.ndarray:
    """His gate for one population: the events inside the convex hull of the events that
    bin_mask keeps, tested against a triangulation of that hull's vertices rather than of every
    kept event. Empty when the cells or the kept events can't make a hull."""
    raw = np.column_stack((np.asarray(fsc_a, dtype=float), np.asarray(ssc_a, dtype=float)))
    points = raw[bin_mask(zz, cells)]
    if len(points) <= MIN_POINTS:
        return np.zeros(len(raw), dtype=bool)
    try:
        hull = ConvexHull(points)
    except QhullError:  # kept events on one line: his Delaunay raises here
        return np.zeros(len(raw), dtype=bool)
    return _inside(points[hull.vertices], raw)


def lymphmacro_finder(fsc_a, ssc_a, *, n_row=11, n_col=11, t11=22, t21=0.25, t12=20, t22=0.5):
    """His gate finder: one bool column per population, one row per event, populations in his
    order. The first peak's area takes t11 and t21, the others t12 and t22. No peak left after
    his filters gives zero columns, where his code raises IndexError."""
    return _finder(fsc_a, ssc_a, n_row, n_col, t11, t21, t12, t22, find_area)


def _finder(fsc_a, ssc_a, n_row, n_col, t11, t21, t12, t22, area) -> np.ndarray:
    fsc_a, ssc_a = channel("fsc_a", fsc_a), channel("ssc_a", ssc_a)
    if len(fsc_a) != len(ssc_a):
        raise ValueError(f"fsc_a and ssc_a must have the same length, got {len(fsc_a)} and {len(ssc_a)}")
    grid = f_plot(*sqrt_coords(fsc_a, ssc_a), nr=MSIZE, nc=MSIZE, scale="l")
    x = grid.z.T
    peaks = refine_peaks(x, local_maxima(x, n_row=n_row, n_col=n_col))
    masks = np.zeros((len(fsc_a), len(peaks)), dtype=bool)
    for k, (i, j) in enumerate(peaks):
        t1, t2 = (t11, t21) if k == 0 else (t12, t22)
        masks[:, k] = hull_gate(fsc_a, ssc_a, grid.zz, area(x, i, j, t1, t2))
    return masks
```

```python
from __future__ import annotations

import matplotlib.pyplot as plt
import numpy as np

# his scatter sizes in points squared, drawn as markers of that area: ms = sqrt(s)
EVENT_SIZE = 0.5
POP_SIZE = 5.0
EVENT_ALPHA = 0.3
POP_ALPHA = 0.8
EDGE_WIDTH = 1.0  # a scatter marker's edge, patch.linewidth


def gates(fsc_a, ssc_a, masks, *, title=None, ax=None):
    """His figure: the events in gray, then each population with events, in the color cycle's
    next color, his legend at markerscale 4. Each layer is one Line2D of markers, which draws
    the same picture several times faster than scatter, rasterized so a PDF stays small."""
    fsc = np.asarray(fsc_a, dtype=float)
    ssc = np.asarray(ssc_a, dtype=float)
    m = np.asarray(masks)
    if fsc.ndim != 1 or ssc.shape != fsc.shape:

    if m.dtype != bool or m.ndim != 2:

        raise ValueError(f"masks must have one row per event, {len(fsc)}, got {len(m)}")
    ax = plt.gca() if ax is None else ax
    style = {"ls": "none", "mew": EDGE_WIDTH, "rasterized": True}
```
